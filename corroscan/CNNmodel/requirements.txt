# CorroScan CNN — Python dependencies
# Install with:  pip install -r requirements.txt

torch           # deep learning framework (CNN training + inference)
torchvision     # pretrained EfficientNet-B0 model + image transforms
numpy           # array operations
opencv-python   # image processing: masking, morphology, corrosion segmentation
pillow          # image loading for PyTorch datasets
pandas          # loading the labels CSV
scikit-learn    # train/test split, classification metrics
matplotlib      # training history plots (train.py)
scipy           # gaussian_kde for pit depth probability distribution
flask           # REST API server
flask-cors      # cross-origin requests from the React frontend
